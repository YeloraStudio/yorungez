cmake_minimum_required(VERSION 3.22)
project(yorungez_sky C CXX)
add_library(yorungez_sky SHARED yorungez_sky.c
    ../../../third-party/astronomy-engine/source/c/astronomy.c)
target_compile_features(yorungez_sky PRIVATE c_std_99)
target_compile_options(yorungez_sky PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(yorungez_sky m)
if(ANDROID)
    target_link_options(yorungez_sky PRIVATE "-Wl,-z,max-page-size=16384")
endif()
